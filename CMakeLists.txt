cmake_minimum_required(VERSION 3.20)
project(command_journal VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow -Werror)
if(NOT UNIX)
  message(FATAL_ERROR "command-journal requires POSIX file operations and flock")
endif()
option(JOURNAL_SANITIZERS "Enable address and undefined-behavior checks" OFF)
add_library(journal src/codec.cpp src/executor.cpp)
target_include_directories(journal PUBLIC include)
target_compile_features(journal PUBLIC cxx_std_20)
if(JOURNAL_SANITIZERS)
  target_compile_options(journal PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(journal PUBLIC -fsanitize=address,undefined)
endif()
add_executable(command-journal src/main.cpp)
target_link_libraries(command-journal PRIVATE journal)
enable_testing()
add_executable(journal-tests tests/test_journal.cpp)
target_link_libraries(journal-tests PRIVATE journal)
add_test(NAME codec-and-recovery COMMAND journal-tests $<TARGET_FILE:command-journal>)
